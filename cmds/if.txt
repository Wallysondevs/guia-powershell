# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Erros de SINTAXE (#ERRO-PARSE) precisam rodar isolados (Invoke-Expression), senão derrubam o arquivo todo.
# Só linguagem: nenhuma lista cria nem apaga arquivo, não precisa de pasta de teste.
# shell: pwsh
## o if simples
$nota = 8
if ($nota -ge 7) { "Aprovada" }
$nota = 5
if ($nota -ge 7) { "Aprovada" }
"(nada apareceu)"
## a condição vira verdadeiro ou falso
$nota -ge 7
8 -ge 7
## valores que contam como falso
if (0) { "sim" } else { "0 é falso" }
if ("") { "sim" } else { "texto vazio é falso" }
if ($null) { "sim" } else { "$null é falso" }
if ("0") { "texto '0' é verdadeiro" }
if (@()) { "sim" } else { "lista vazia é falsa" }
## várias linhas no bloco
$estoque = 3
if ($estoque -lt 5) {
    Write-Output "Estoque baixo: $estoque unidades"
    Write-Output "Hora de comprar mais"
}
## erros
if ($nota = 10) { "sempre entra: isso é atribuição" }; "nota agora vale $nota"
Invoke-Expression 'if ($nota -ge 7) "Aprovada"'   #ERRO-PARSE
Invoke-Expression 'if $nota -ge 7 { "Aprovada" }'   #ERRO-PARSE
