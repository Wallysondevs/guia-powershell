# operadores-colecao — comandos para colher no Windows (só leitura/demonstração)
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Anonimizar antes de usar.
# shell: pwsh
## -contains: a lista é quem contém
"Ana", "Beto", "Caio" -contains "Beto"
"Ana", "Beto", "Caio" -contains "beto"
"Ana", "Beto", "Caio" -ccontains "beto"
"Ana", "Beto", "Caio" -notcontains "Dora"
## -in: o valor está na lista
"Beto" -in "Ana", "Beto", "Caio"
"Dora" -notin "Ana", "Beto", "Caio"
2 -in 1, 2, 3
## não é "está dentro do texto"
"a" -in "abc"
"abc" -contains "a"
## lista vazia
@() -contains "Ana"
$null -in @()
