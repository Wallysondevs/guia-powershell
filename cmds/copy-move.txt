# copy-move — comandos para colher no Windows (só leitura/demonstração)
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Anonimizar antes de usar.
# shell: pwsh
## preparar pasta de teste
$T = Join-Path $HOME 'guia-ps-teste'
New-Item -ItemType Directory -Path $T -Force | Out-Null
Set-Location $T
# shell: pwsh
New-Item -ItemType Directory backup, docs | Out-Null
'oi' | Set-Content nota.txt
'x' | Set-Content docs\a.txt
## copiar
Copy-Item nota.txt backup\
Copy-Item nota.txt nota-copia.txt
Copy-Item docs backup\docs-copia
Get-ChildItem backup -Recurse -Name
Copy-Item docs backup\docs-rec -Recurse
Get-ChildItem backup -Recurse -Name
Copy-Item nota.txt backup\ -PassThru
## mover
Move-Item nota-copia.txt docs\
Get-ChildItem docs -Name
Move-Item docs\nota-copia.txt .\renomeada.txt
Get-ChildItem -Name
## erros
Copy-Item naoexiste.txt backup\   #ERRO
Move-Item renomeada.txt backup\nota.txt   #ERRO
Copy-Item nota.txt backup\ -WhatIf
## limpar (sempre no fim)
Set-Location $HOME
Remove-Item -Recurse -Force (Join-Path $HOME 'guia-ps-teste')
Test-Path (Join-Path $HOME 'guia-ps-teste')
