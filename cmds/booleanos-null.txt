# booleanos-null — comandos para colher no Windows (só leitura/demonstração)
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Anonimizar antes de usar.
# shell: pwsh
## os três valores
$true
$false
$null
$true.GetType().Name
## comparar com $null (o $null vai à esquerda)
$a = $null
$null -eq $a
$null -ne "Ana"
## o que o if considera falso
if ($true) { "verdadeiro" } else { "falso" }
if ($false) { "verdadeiro" } else { "falso" }
if ($null) { "verdadeiro" } else { "falso" }
if (0) { "verdadeiro" } else { "falso" }
if ("") { "verdadeiro" } else { "falso" }
if (@()) { "verdadeiro" } else { "falso" }
if ("Ana") { "verdadeiro" } else { "falso" }
## operador ?? (só no 7): se a esquerda for $null, usa a direita
$a = $null
$a ?? "padrão"
"já tem" ?? "padrão"
# shell: powershell
## no 5.1 o ?? não existe
$a = $null
$a ?? "padrão"   #ERRO
# shell: pwsh
