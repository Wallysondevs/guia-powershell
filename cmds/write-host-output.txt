# write-host-output — autor Tito
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Erros de SINTAXE (#ERRO-PARSE) precisam rodar isolados (Invoke-Expression), senão derrubam o arquivo todo.
# Só linguagem: nenhuma lista cria nem apaga arquivo, não precisa de pasta de teste.
# shell: pwsh
## os dois na tela
Write-Output "Olá, Ana!"
Write-Host "Olá, Ana!"
"Olá, Ana!"
## a diferença: quem segue pela esteira
Write-Output "maçã","banana","uva" | Measure-Object
Write-Host "maçã","banana","uva" | Measure-Object
$texto = Write-Output "guardado"; "Na variável: [$texto]"
$texto = Write-Host "não guardado"; "Na variável: [$texto]"
## cores e separadores (só Write-Host)
Write-Host "Atenção" -ForegroundColor Yellow
Write-Host "a","b","c" -Separator " | "
Write-Host "sem quebra " -NoNewline; Write-Host "de linha"
## erro
Write-Output "oi" -ForegroundColor Red   #ERRO
Write-Host "oi" -ForegroundColor Roxo   #ERRO
