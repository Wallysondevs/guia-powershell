# conversao — comandos para colher no Windows (só leitura/demonstração)
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Anonimizar antes de usar.
# shell: pwsh
## conversão com colchetes
[int]"42"
[string]42
[double]"3,14"
[double]"3.14"
[datetime]"2026-11-20"
[int]3.9
## o operador -as (se não der, volta $null, sem erro)
"42" -as [int]
"abc" -as [int]
$null -eq ("abc" -as [int])
## o operador -is
42 -is [int]
"42" -is [int]
"42" -is [string]
(1, 2) -is [array]
## a pegadinha do [bool]
[bool]0
[bool]1
[bool]""
[bool]"false"
[bool]$null
## erro
[int]"abc"   #ERRO
[datetime]"não é data"   #ERRO
