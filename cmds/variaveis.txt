# variaveis — comandos para colher no Windows (só leitura/demonstração)
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Anonimizar antes de usar.
# shell: pwsh
## criar e ver
$nome = "Ana"
$nome
$idade = 30
$idade
## trocar o valor
$nome = "Beto"
$nome
## usar no texto e em conta
"$nome tem $idade anos"
$idade + 1
## nomes com chaves
${cidade} = "Recife"
${cidade}
Get-Variable nome, idade, cidade | Format-Table Name, Value -AutoSize

Remove-Variable idade
