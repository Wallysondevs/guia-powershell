# foreach — autor Tito
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Erros de SINTAXE (#ERRO-PARSE) precisam rodar isolados (Invoke-Expression), senão derrubam o arquivo todo.
# Só linguagem: nenhuma lista cria nem apaga arquivo, não precisa de pasta de teste.
# shell: pwsh
## básico
$frutas = "maçã", "banana", "uva"
foreach ($fruta in $frutas) { "Eu gosto de $fruta" }
## somar
$gastos = 12.5, 30, 7.25
$total = 0
foreach ($g in $gastos) { $total += $g }
"Total: $total"
## com intervalo
foreach ($n in 1..5) { "Linha $n" }
## guardar o resultado
$dobro = foreach ($n in 1..3) { $n * 2 }
$dobro -join ", "
## foreach x ForEach-Object
1..3 | ForEach-Object { $_ * 10 }
## lista vazia e $null
foreach ($x in @()) { "nunca" }; "lista vazia: nada rodou"
foreach ($x in $null) { "rodou com [$x]" }
## erros
Invoke-Expression 'foreach ($fruta of $frutas) { $fruta }'   #ERRO-PARSE
Invoke-Expression 'foreach $fruta in $frutas { $fruta }'   #ERRO-PARSE
