# rename-item — comandos para colher no Windows (só leitura/demonstração)
# ATENÇÃO 20/11: no Windows só existe o 5.1 (sem pwsh). Blocos que comparam com o 7 ficam ilustrativos ou vêm de outra fonte.
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Anonimizar antes de usar.
# shell: powershell
## preparar pasta de teste
$T = Join-Path $HOME 'guia-ps-teste'
New-Item -ItemType Directory -Path $T -Force | Out-Null
Set-Location $T
# shell: powershell
'oi' | Set-Content nota.txt
1..3 | ForEach-Object { New-Item "foto$_.JPG" | Out-Null }
## renomear
Rename-Item nota.txt anotacao.txt
Get-ChildItem -Name
Rename-Item anotacao.txt -NewName lembrete.txt -PassThru
## vários de uma vez
Get-ChildItem *.JPG | Rename-Item -NewName { $_.Name -replace '\.JPG$', '.jpg' } -WhatIf
Get-ChildItem *.JPG | Rename-Item -NewName { $_.Name -replace '\.JPG$', '.jpg' }
Get-ChildItem -Name
## erros
Rename-Item naoexiste.txt x.txt   #ERRO
Rename-Item lembrete.txt foto1.jpg   #ERRO
Rename-Item lembrete.txt .\pasta\x.txt   #ERRO
## limpar (sempre no fim)
Set-Location $HOME
Remove-Item -Recurse -Force (Join-Path $HOME 'guia-ps-teste')
Test-Path (Join-Path $HOME 'guia-ps-teste')
