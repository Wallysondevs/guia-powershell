# introducao — comandos para colher no Windows (só leitura/demonstração)
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Anonimizar antes de usar.
# shell: pwsh
## o PowerShell responde com objetos
Get-Date
(Get-Date).Year
Get-Date | Get-Member -MemberType Property | Select-Object -First 5
## comparação com o cmd
cmd /c echo %DATE%
