# variaveis-automaticas — comandos para colher no Windows (só leitura/demonstração)
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Anonimizar antes de usar.
# shell: pwsh
## pasta do usuário e pasta atual
$HOME
$PWD
## versão e edição
$PSVersionTable
$PSEdition
$PSVersionTable.PSVersion
## Windows, Linux ou macOS (só no 7)
$IsWindows
$IsLinux
$IsMacOS
## deu certo? ($?)
$?
Get-Date | Out-Null
$?
Get-Variavel   #ERRO
$?
## o $_ da esteira
1, 2, 3 | ForEach-Object { $_ * 10 }
## outras úteis
$PID
$Host.Name
# shell: powershell
## no 5.1: edição Desktop e sem $IsWindows
$PSVersionTable
$PSEdition
Get-Variable IsWindows, IsLinux, IsMacOS -ErrorAction SilentlyContinue
# shell: pwsh
