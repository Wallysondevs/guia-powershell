# provedores — comandos para colher no Windows (só leitura/demonstração)
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Anonimizar antes de usar.
# shell: pwsh
## drives e provedores (só leitura)
Get-PSProvider
Get-PSDrive | Format-Table Name, Provider, Root -AutoSize   # anonimizar tamanhos/rede se aparecer
Get-Item Env:OS, Env:USERNAME, Env:windir   # NUNCA listar Env: aberto
$env:OS
Get-Item Env:COMPUTERNAME   # vira NOTEBOOK
Get-ChildItem HKCU:\ -Name
Get-ItemProperty 'HKCU:\Control Panel\International' -Name LocaleName, sShortDate
Get-ChildItem Function: | Select-Object -First 5
Get-ChildItem Alias: | Measure-Object
Get-ChildItem Variable: | Select-Object -First 5
Get-ChildItem Cert:\CurrentUser -Name
## erro
Get-ChildItem Z:\   #ERRO
