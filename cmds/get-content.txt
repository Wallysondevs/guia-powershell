# get-content — comandos para colher no Windows (só leitura/demonstração)
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Anonimizar antes de usar.
# shell: pwsh
## preparar pasta de teste
$T = Join-Path $HOME 'guia-ps-teste'
New-Item -ItemType Directory -Path $T -Force | Out-Null
Set-Location $T
# shell: pwsh
1..12 | ForEach-Object { "Linha $_" } | Set-Content linhas.txt
"nome,idade`nAna,30`nBeto,25" | Set-Content pessoas.csv
## ler
Get-Content linhas.txt
Get-Content linhas.txt -TotalCount 3
Get-Content linhas.txt -Tail 2
(Get-Content linhas.txt).Count
(Get-Content linhas.txt)[0]
Get-Content linhas.txt -Raw | Measure-Object -Character
cat pessoas.csv
Get-Content pessoas.csv | Select-String Ana
## erro
Get-Content naoexiste.txt   #ERRO
## limpar (sempre no fim)
Set-Location $HOME
Remove-Item -Recurse -Force (Join-Path $HOME 'guia-ps-teste')
Test-Path (Join-Path $HOME 'guia-ps-teste')
