# break-continue — autor Tito
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Erros de SINTAXE (#ERRO-PARSE) precisam rodar isolados (Invoke-Expression), senão derrubam o arquivo todo.
# Só linguagem: nenhuma lista cria nem apaga arquivo, não precisa de pasta de teste.
# shell: pwsh
## break
foreach ($n in 1..10) { if ($n -eq 4) { break }; "n = $n" }
"depois do laço"
## continue
foreach ($n in 1..6) { if ($n % 2 -eq 0) { continue }; "ímpar: $n" }
## procurar e parar
$notas = 6, 8, 3, 9
foreach ($nota in $notas) { if ($nota -lt 5) { "Achei nota baixa: $nota"; break } }
## laço dentro de laço e rótulo
foreach ($i in 1..3) { foreach ($j in 1..3) { if ($j -eq 2) { break }; "$i-$j" } }
:fora foreach ($i in 1..3) { foreach ($j in 1..3) { if ($j -eq 2) { break fora }; "$i-$j" } }
## a pegadinha do ForEach-Object
1..5 | ForEach-Object { if ($_ -eq 3) { return }; "item $_" }
& { 1..5 | ForEach-Object { if ($_ -eq 3) { break }; "item $_" }; "esta linha some" }; "fora do bloco"
