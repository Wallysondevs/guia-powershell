# versoes-powershell — comandos para colher no Windows (só leitura/demonstração)
# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Anonimizar antes de usar.
# shell: powershell
## Windows PowerShell 5.1
$PSVersionTable
$PSVersionTable.PSEdition
(Get-Command powershell).Source
Get-ExecutionPolicy -List
# shell: pwsh
## PowerShell 7
$PSVersionTable
$PSVersionTable.PSEdition
(Get-Command pwsh).Source
Get-ExecutionPolicy -List
## os dois lado a lado
Get-Command pwsh, powershell -ErrorAction SilentlyContinue | Format-Table Name, Version, Source
