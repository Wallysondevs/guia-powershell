# Formato: "## rótulo" = nome do bloco na aula; "# shell: pwsh|powershell" vale até o próximo "# shell:".
# Linhas com "#ERRO" são erros de propósito (precisamos da mensagem real). Erros de SINTAXE (#ERRO-PARSE) precisam rodar isolados (Invoke-Expression), senão derrubam o arquivo todo.
# Só linguagem: nenhuma lista cria nem apaga arquivo, não precisa de pasta de teste.
# shell: pwsh
## básico
for ($i = 1; $i -le 5; $i++) { "Volta $i" }
## de 2 em 2 e de trás pra frente
for ($i = 0; $i -le 10; $i += 2) { $i }
for ($i = 3; $i -ge 1; $i--) { "$i..." }; "Já!"
## com índice da lista
$alunos = "Ana", "Beto", "Caio"
for ($i = 0; $i -lt $alunos.Count; $i++) { "$($i + 1). $($alunos[$i])" }
## erros
$alunos[3]
$null -eq $alunos[3]
Set-StrictMode -Version Latest; $alunos[3]   #ERRO
Set-StrictMode -Off
Invoke-Expression 'for ($i = 1, $i -le 3, $i++) { $i }'   #ERRO-PARSE
for ($i = 1; $i -le 3; $i--) { $i; if ($i -lt -2) { "(parei o laço infinito na mão)"; break } }
